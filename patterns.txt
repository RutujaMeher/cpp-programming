1. Square Star Pattern

#include <iostream>
using namespace std;

int main() {
    for (int i = 1; i <= 5; i++) {
        for (int j = 1; j <= 5; j++) {
            cout << "* ";
        }
        cout << endl;
    }

    return 0;
}



2. Increasing Star Triangle

#include <iostream>
using namespace std;

int main() {
    for (int i = 1; i <= 5; i++) {
        for (int j = 1; j <= i; j++) {
            cout << "* ";
        }
        cout << endl;
    }

    return 0;
}



3. Decreasing Star Triangle

#include <iostream>
using namespace std;

int main() {
    for (int i = 5; i >= 1; i--) {
        for (int j = 1; j <= i; j++) {
            cout << "* ";
        }
        cout << endl;
    }

    return 0;
}



4. Right Aligned Triangle

#include <iostream>
using namespace std;

int main() {
    for (int i = 1; i <= 5; i++) {

        for (int j = 1; j <= 5 - i; j++) {
            cout << "  ";
        }

        for (int j = 1; j <= i; j++) {
            cout << "* ";
        }

        cout << endl;
    }

    return 0;
}

7. Pyramid Pattern

#include <iostream>
using namespace std;

int main() {
    for (int i = 1; i <= 5; i++) {

        for (int j = 1; j <= 5 - i; j++) {
            cout << " ";
        }

        for (int j = 1; j <= 2 * i - 1; j++) {
            cout << "*";
        }

        cout << endl;
    }

    return 0;
}



8. Inverted Pyramid

#include <iostream>
using namespace std;

int main() {
    for (int i = 5; i >= 1; i--) {

        for (int j = 1; j <= 5 - i; j++) {
            cout << " ";
        }

        for (int j = 1; j <= 2 * i - 1; j++) {
            cout << "*";
        }

        cout << endl;
    }

    return 0;
}



9. Continuous Number Triangle

#include <iostream>
using namespace std;

int main() {
    int num = 1;

    for (int i = 1; i <= 5; i++) {
        for (int j = 1; j <= i; j++) {
            cout << num << " ";
            num++;
        }
        cout << endl;
    }

    return 0;
}



10. Alphabet Triangle

#include <iostream>
using namespace std;

int main() {
    for (int i = 1; i <= 5; i++) {
        for (int j = 1; j <= i; j++) {
            cout << char('A' + j - 1) << " ";
        }
        cout << endl;
    }

    return 0;
}



11. Hollow Square

#include <iostream>
using namespace std;

int main() {
    for (int i = 1; i <= 5; i++) {
        for (int j = 1; j <= 5; j++) {

            if (i == 1 || i == 5 || j == 1 || j == 5)
                cout << "* ";
            else
                cout << "  ";
        }

        cout << endl;
    }

    return 0;
}
