
#include <iostream>
#include <vector>
#include <stack>
#include <queue>
#include <map>
#include <set>
#include <algorithm>
using namespace std;

int main()
{
    // Vector
    vector<int> v = {40, 10, 30, 20};

    v.push_back(50);

    cout << "Vector Elements: ";
    for (int x : v)
        cout << x << " ";

    cout << "\n";

    // Sorting
    sort(v.begin(), v.end());

    cout << "Sorted Vector: ";
    for (int x : v)
        cout << x << " ";

    cout << "\n";

    // Stack
    stack<int> s;

    s.push(10);
    s.push(20);
    s.push(30);

    cout << "Stack Top: " << s.top() << endl;

    s.pop();

    cout << "Stack After Pop: " << s.top() << endl;

    // Queue
    queue<int> q;

    q.push(100);
    q.push(200);
    q.push(300);

    cout << "Queue Front: " << q.front() << endl;

    q.pop();

    cout << "Queue After Pop: " << q.front() << endl;

    // Set
    set<int> st = {10, 20, 10, 30, 40};

    cout << "Set Elements: ";
    for (int x : st)
        cout << x << " ";

    cout << "\n";

    // Map
    map<int, string> students;

    students[1] = "Rahul";
    students[2] = "Amit";
    students[3] = "Priya";

    cout << "Map Elements:\n";

    for (auto x : students)
        cout << x.first << " : " << x.second << endl;

    // Searching
    int key = 30;

    if (find(v.begin(), v.end(), key) != v.end())
        cout << "Element Found" << endl;
    else
        cout << "Element Not Found" << endl;

    return 0;
}
