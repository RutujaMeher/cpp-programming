1. Create and Write a File
#include <iostream>
#include <fstream>
using namespace std;

int main() {
    ofstream fout("data.txt");

    fout << "Hello World";
    fout.close();

    return 0;
}


2. Read a File
#include <iostream>
#include <fstream>
using namespace std;

int main() {
    ifstream fin("data.txt");

    string line;
    while (getline(fin, line)) {
        cout << line << endl;
    }

    fin.close();
    return 0;
}

7. Check Whether File Exists / Opened Successfully
#include <iostream>
#include <fstream>
using namespace std;

int main() {
    ifstream fin("data.txt");

    if (fin.is_open()) {
        cout << "File opened successfully";
    } else {
        cout << "File not found";
    }

    fin.close();
    return 0;
}


8. Read and Write Using fstream
#include <iostream>
#include <fstream>
using namespace std;

int main() {
    fstream file;

    file.open("data.txt", ios::in | ios::out | ios::trunc);

    file << "C++ File Handling";

    file.seekg(0);

    string line;
    getline(file, line);

    cout << line;

    file.close();
    return 0;
}


9. Count Characters in a File
#include <iostream>
#include <fstream>
using namespace std;

int main() {
    ifstream fin("data.txt");

    char ch;
    int count = 0;

    while (fin.get(ch)) {
        count++;
    }

    cout << "Characters: " << count;

    fin.close();
    return 0;
}

11. Copy One File into Another
#include <iostream>
#include <fstream>
using namespace std;

int main() {
    ifstream fin("data.txt");
    ofstream fout("copy.txt");

    char ch;

    while (fin.get(ch)) {
        fout.put(ch);
    }

    fin.close();
    fout.close();

    return 0;
}
